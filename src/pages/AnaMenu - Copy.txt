import { useState, useEffect } from 'react';
import * as firestore from '../firebase/firestore';
import { format, startOfDay } from 'date-fns';
import { tr } from 'date-fns/locale';
import YeniOdeme from '../components/YeniOdeme';
import OdemeListesi from '../components/OdemeListesi';

export default function AnaMenu({ user }) {
  const [tasks, setTasks] = useState({ todo: [], inProgress: [], completed: [] });
  const [draggedItem, setDraggedItem] = useState(null);
  const [dragOverColumn, setDragOverColumn] = useState(null);
  const [showNewPayment, setShowNewPayment] = useState(false);
  const [showListModal, setShowListModal] = useState(false);
  const [selectedTask, setSelectedTask] = useState(null);
  const [selectedReminder, setSelectedReminder] = useState(null);

  const canEdit = user?.role === 'superadmin' || user?.role === 'admin' || user?.role === 'editor';

  useEffect(() => {
    loadTodayTasks();
    window.addEventListener('reminderUpdated', loadTodayTasks);
    return () => window.removeEventListener('reminderUpdated', loadTodayTasks);
  }, []);

  const loadTodayTasks = async () => {
    const today = startOfDay(new Date());
    const todayStr = format(today, 'yyyy-MM-dd');
    
    const [payments, reminders] = await Promise.all([
      firestore.getPayments({ startDate: todayStr, endDate: todayStr }),
      firestore.getReminders()
    ]);

    const isAdmin = user?.role === 'superadmin' || user?.role === 'admin';
    const filteredPayments = isAdmin ? payments : payments.filter(p => !p.is_admin_only);

    const todo = [];
    const completed = [];

    filteredPayments.forEach(payment => {
      if (payment.payment_method === 'devir') return;
      
      const task = {
        id: payment.id,
        type: 'payment',
        title: payment.payment_type === 'kredi_karti' 
          ? `${payment.credit_card_name || 'Kart'} Ödemesi`
          : payment.payment_type === 'cari'
          ? `${payment.cari_name || 'Cari'} Ödemesi`
          : 'Serbest Ödeme',
        amount: payment.amount,
        description: payment.description,
        time: payment.payment_date,
        data: payment
      };

      if (payment.is_completed) {
        completed.push(task);
      } else {
        todo.push(task);
      }
    });

    const currentDay = today.getDate();
    const [cards, cariList] = await Promise.all([
      firestore.getCreditCards(),
      firestore.getCari()
    ]);

    reminders.forEach(reminder => {
      if (reminder.isActive === false) return;
      if (!reminder.remainingCount || reminder.remainingCount <= 0) return;

      const dayStart = parseInt(reminder.dayStart) || 0;
      const dayEnd = parseInt(reminder.dayEnd) || 0;

      let isInRange = false;
      if (dayStart && dayEnd) {
        if (dayStart <= dayEnd) {
          isInRange = currentDay >= dayStart && currentDay <= dayEnd;
        } else {
          isInRange = currentDay >= dayStart || currentDay <= dayEnd;
        }
      }

      if (isInRange) {
        let title = reminder.title;
        if (reminder.type === 'creditCard') {
          const card = cards.find(c => c.id === reminder.creditCardId);
          title = `${card?.owner_name || card?.name || 'Kart'} Ödemesi`;
        } else if (reminder.type === 'cari') {
          const cari = cariList.find(c => c.id === reminder.cariId);
          title = `${cari?.name || 'Cari'} Ödemesi`;
        }

        const task = {
          id: reminder.id,
          type: 'reminder',
          title,
          description: `Gün: ${dayStart}-${dayEnd} | Kalan: ${reminder.remainingCount}/${reminder.paymentCount}`,
          time: format(today, 'yyyy-MM-dd'),
          data: reminder
        };
        todo.push(task);
      }
    });

    setTasks({ todo, inProgress: [], completed });
  };

  const handleDragStart = (e, item, column) => {
    setDraggedItem({ item, column });
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', item.id);
  };

  const handleDragOver = (e, column) => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    setDragOverColumn(column);
  };

  const handleDragLeave = () => {
    setDragOverColumn(null);
  };

  const handleDrop = async (e, targetColumn) => {
    e.preventDefault();
    setDragOverColumn(null);
    if (!draggedItem) return;

    const { item, column: sourceColumn } = draggedItem;

    if (sourceColumn === targetColumn) {
      setDraggedItem(null);
      return;
    }

    if (item.type === 'payment' && targetColumn === 'completed') {
      await firestore.updatePayment(item.id, { is_completed: true });
      item.data.is_completed = true;
    } else if (item.type === 'payment' && sourceColumn === 'completed') {
      await firestore.updatePayment(item.id, { is_completed: false });
      item.data.is_completed = false;
    }

    const newTasks = { ...tasks };
    newTasks[sourceColumn] = newTasks[sourceColumn].filter(t => t.id !== item.id);
    newTasks[targetColumn] = [...newTasks[targetColumn], item];
    setTasks(newTasks);
    setDraggedItem(null);
  };

  const handleTaskClick = (task) => {
    if (task.type === 'payment') {
      setSelectedTask(task);
      setShowListModal(true);
    } else if (task.type === 'reminder' && canEdit) {
      setSelectedReminder(task.data);
      setShowNewPayment(true);
    }
  };

  const TaskCard = ({ task, column }) => {
    const isDragging = draggedItem?.item.id === task.id;
    
    return (
      <div
        draggable
        onDragStart={(e) => handleDragStart(e, task, column)}
        onDragEnd={() => setDraggedItem(null)}
        onClick={() => handleTaskClick(task)}
        className={`bg-white rounded-lg p-4 mb-3 shadow-sm border-2 transition-all cursor-pointer ${
          isDragging 
            ? 'opacity-50 border-blue-400' 
            : 'border-gray-200 hover:shadow-lg hover:border-blue-300'
        }`}
      >
        <div className="flex items-start justify-between mb-2">
          <h4 className="font-semibold text-gray-800 text-sm flex-1">{task.title}</h4>
          {task.type === 'payment' && (
            <span className="px-2 py-1 bg-blue-100 text-blue-700 rounded text-xs font-semibold ml-2">
              Ödeme
            </span>
          )}
          {task.type === 'reminder' && (
            <span className="px-2 py-1 bg-purple-100 text-purple-700 rounded text-xs font-semibold ml-2">
              Hatırlatma
            </span>
          )}
        </div>
        {task.description && (
          <p className="text-xs text-gray-600 mb-2">{task.description}</p>
        )}
        {task.amount && (
          <p className="text-sm font-bold text-red-600">
            {task.amount.toLocaleString('tr-TR', { minimumFractionDigits: 2 })} ₺
          </p>
        )}
      </div>
    );
  };

  const Column = ({ title, tasks: columnTasks, column, icon, color }) => {
    const isOver = dragOverColumn === column;
    return (
      <div className={`flex-1 bg-gray-50 rounded-xl p-4 border-2 transition-all ${
        isOver ? 'border-blue-400 bg-blue-50' : 'border-gray-200'
      }`}>
        <div className="flex items-center gap-2 mb-4">
          {icon}
          <h3 className="font-bold text-gray-800 text-lg">{title}</h3>
          <span className={`ml-auto px-3 py-1 ${color} rounded-full text-sm font-semibold`}>
            {columnTasks.length}
          </span>
        </div>
        <div
          onDragOver={(e) => handleDragOver(e, column)}
          onDragLeave={handleDragLeave}
          onDrop={(e) => handleDrop(e, column)}
          className="min-h-[500px] transition-colors"
        >
          {columnTasks.length === 0 ? (
            <div className="flex flex-col items-center justify-center py-12 text-gray-400">
              <svg className="w-16 h-16 mb-3 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12h6m-6 4h6m2 5H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z" />
              </svg>
              <p className="text-sm font-medium">Hiç görev yok</p>
            </div>
          ) : (
            columnTasks.map(task => (
              <TaskCard key={task.id} task={task} column={column} />
            ))
          )}
        </div>
      </div>
    );
  };

  return (
    <div className="w-full h-full bg-gradient-to-br from-gray-50 to-blue-50 overflow-auto p-6">
      <div className="max-w-[1400px] mx-auto">
        <div className="bg-white rounded-2xl shadow-lg p-6 mb-6">
          <div className="flex items-center justify-between">
            <div>
              <h1 className="text-3xl font-bold text-gray-900">Bugünün Görevleri</h1>
              <p className="text-gray-600 mt-1">{format(new Date(), 'dd MMMM yyyy, EEEE', { locale: tr })}</p>
            </div>
            <div className="flex items-center gap-4">
              <div className="text-center px-4 py-2 bg-blue-50 rounded-lg">
                <p className="text-xs text-gray-600 mb-1">Toplam</p>
                <p className="text-2xl font-bold text-blue-600">{tasks.todo.length + tasks.inProgress.length + tasks.completed.length}</p>
              </div>
              <div className="text-center px-4 py-2 bg-green-50 rounded-lg">
                <p className="text-xs text-gray-600 mb-1">Tamamlanan</p>
                <p className="text-2xl font-bold text-green-600">{tasks.completed.length}</p>
              </div>
              {canEdit && (
              <button
                onClick={() => {
                  setSelectedReminder(null);
                  setShowNewPayment(true);
                }}
                className="px-6 py-3 bg-gradient-to-r from-blue-600 to-blue-700 text-white rounded-xl hover:from-blue-700 hover:to-blue-800 transition-all font-semibold shadow-lg flex items-center gap-2"
              >
                <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4" />
                </svg>
                Yeni Ödeme
              </button>
              )}
            </div>
          </div>
        </div>

        <div className="grid grid-cols-3 gap-6">
          <Column
            title="Yapılacaklar"
            tasks={tasks.todo}
            column="todo"
            color="bg-blue-100 text-blue-700"
            icon={
              <svg className="w-6 h-6 text-blue-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5H7a2 2 0 00-2 2v12a2 2 0 002 2h10a2 2 0 002-2V7a2 2 0 00-2-2h-2M9 5a2 2 0 002 2h2a2 2 0 002-2M9 5a2 2 0 012-2h2a2 2 0 012 2" />
              </svg>
            }
          />
          <Column
            title="Beklemede"
            tasks={tasks.inProgress}
            column="inProgress"
            color="bg-yellow-100 text-yellow-700"
            icon={
              <svg className="w-6 h-6 text-yellow-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z" />
              </svg>
            }
          />
          <Column
            title="Tamamlananlar"
            tasks={tasks.completed}
            column="completed"
            color="bg-green-100 text-green-700"
            icon={
              <svg className="w-6 h-6 text-green-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z" />
              </svg>
            }
          />
        </div>
      </div>

      {showNewPayment && (
        <YeniOdeme
          selectedDate={new Date()}
          onClose={() => {
            setShowNewPayment(false);
            setSelectedReminder(null);
            loadTodayTasks();
            window.dispatchEvent(new Event('reminderUpdated'));
          }}
          preSelectedCard={selectedReminder?.type === 'creditCard' ? selectedReminder.creditCardId : null}
          preSelectedCari={selectedReminder?.type === 'cari' ? selectedReminder.cariId : null}
        />
      )}

      {showListModal && selectedTask && (
        <OdemeListesi
          selectedDate={new Date(selectedTask.time)}
          payments={[selectedTask.data]}
          onClose={() => {
            setShowListModal(false);
            setSelectedTask(null);
            loadTodayTasks();
          }}
          canEdit={canEdit}
        />
      )}
    </div>
  );
}
